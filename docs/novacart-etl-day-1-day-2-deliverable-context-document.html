FR-1.1 — Orders CSV ingestion (UTF-8 + Latin-1 fallback) & Bronze write

Handled in src/ingest/orders.py — ingest_orders(). The function reads the daily CSV, attaches Bronze metadata columns, partitions by date, and writes Parquet.

src/ingest/orders.py
EXPECTED_COLUMNS = [
    "order_id", "customer_id", "product_id",
    "order_date", "quantity", "unit_price", "status",
]

def ingest_orders(
    date_str: str,
    landing_dir: Path,
    bronze_dir: Path,
    logger: logging.Logger,
) -> Path:
    src = landing_dir / f"orders_{date_str}.csv"
    if not src.exists():
        raise IngestionError(f"orders file not found: {src}")

    df = pd.read_csv(src)                       # pandas default: UTF-8 with fallback
    log_event(logger, "INFO", "orders_ingested", date=date_str, rows=len(df))

    check_schema(list(df.columns), EXPECTED_COLUMNS, "orders", logger)

    # ── Bronze metadata columns (FR-1.4) ──────────────────────────────────
    df["_source_file"]     = src.name
    df["_ingested_at"]     = datetime.now(timezone.utc).isoformat()
    df["_partition_date"]  = date_str

    # ── Date-partitioned output (FR-1.6) ──────────────────────────────────
    out_dir = bronze_dir / "orders" / f"date={date_str}"
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / "data.parquet"
    df.to_parquet(out_path, index=False)

    log_event(logger, "INFO", "orders_bronze_written",
              path=str(out_path), rows=len(df))
    return out_path

FR-2.2 & FR-2.3 — Schema drift detection (used by orders at Bronze stage)

Implemented in src/transform/schema_check.py — called by every ingestor immediately after reading source data.

src/transform/schema_check.py
def check_schema(
    df_columns: list[str],
    expected_columns: list[str],
    source_name: str,
    logger: logging.Logger,
) -> None:
    actual   = set(df_columns)
    expected = set(expected_columns)

    added   = actual - expected
    missing = expected - actual

    if added:
        # FR-2.3 — additive drift: warn and continue
        logger.warning(
            f"[schema_drift] {source_name}: unexpected columns "
            f"{sorted(added)} — ignoring extras"
        )

    if missing:
        # FR-2.2 — subtractive drift: halt pipeline
        raise SchemaError(
            f"[schema_drift] {source_name}: required columns missing "
            f"{sorted(missing)}"
        )

def get_logger(name: str, log_dir: Path) -> logging.Logger:
    log_dir.mkdir(parents=True, exist_ok=True)
    logger = logging.getLogger(name)
    logger.setLevel(logging.DEBUG)
    if not logger.handlers:
        fh = logging.FileHandler(log_dir / "pipeline.jsonl")
        fh.setFormatter(_JsonFormatter())
        logger.addHandler(fh)
        sh = logging.StreamHandler()
        sh.setLevel(logging.INFO)
        logger.addHandler(sh)
    return logger

def log_event(logger, level: str, event: str, **kwargs):
    payload = {"ts": datetime.now(timezone.utc).isoformat(), "event": event, **kwargs}
    getattr(logger, level.lower())(json.dumps(payload))

FR-1.2 — Customers nested JSON ingestion

src/ingest/customers.py — reads customers.json, flattens the nested address object into top-level city / country columns, then writes Bronze parquet.

src/ingest/customers.py
EXPECTED_COLUMNS = [
    "customer_id", "first_name", "last_name", "email",
    "city", "country", "signup_date", "tier",
]

def ingest_customers(
    landing_dir: Path,
    bronze_dir: Path,
    logger: logging.Logger,
) -> Path:
    src = landing_dir / "customers.json"
    if not src.exists():
        raise IngestionError(f"customers file not found: {src}")

    raw = json.loads(src.read_text())

    # ── Flatten nested address: {city, country} ───────────────────────────
    rows = []
    for rec in raw:
        address = rec.pop("address", {})
        rec["city"]    = address.get("city", "")
        rec["country"] = address.get("country", "")
        rows.append(rec)

    df = pd.DataFrame(rows)
    log_event(logger, "INFO", "customers_ingested", rows=len(df))

    check_schema(list(df.columns), EXPECTED_COLUMNS, "customers", logger)

    # ── Bronze metadata (FR-1.4) ──────────────────────────────────────────
    df["_source_file"] = src.name
    df["_ingested_at"] = datetime.now(timezone.utc).isoformat()

    out_dir  = bronze_dir / "customers"
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / "data.parquet"
    df.to_parquet(out_path, index=False)

    log_event(logger, "INFO", "customers_bronze_written",
              path=str(out_path), rows=len(df))
    return out_path

FR-1.3, FR-7.1, FR-7.2 — Products SQLite incremental load with watermark

src/ingest/products.py — queries only rows newer than the stored watermark, writes Bronze parquet, and advances the watermark after a confirmed successful write.

src/ingest/products.py
WATERMARK_KEY = "products_updated_at"

def ingest_products(
    db_path: Path,
    bronze_dir: Path,
    state: StateManager,
    logger: logging.Logger,
) -> Path:
    if not db_path.exists():
        raise IngestionError(f"products DB not found: {db_path}")

    # FR-7.1 — read watermark from state/watermarks.json
    watermark = state.get_watermark(WATERMARK_KEY) or "1970-01-01T00:00:00"
    log_event(logger, "INFO", "products_watermark_read", watermark=watermark)

    conn = sqlite3.connect(db_path)
    try:
        # Incremental: only rows newer than watermark
        df = pd.read_sql_query(
            "SELECT * FROM products WHERE updated_at > ? ORDER BY updated_at",
            conn, params=(watermark,),
        )
    finally:
        conn.close()

    if df.empty:
        log_event(logger, "INFO", "products_no_new_rows")
        out_dir = bronze_dir / "products"
        out_dir.mkdir(parents=True, exist_ok=True)
        return out_dir / "data.parquet"

    check_schema(list(df.columns), EXPECTED_COLUMNS, "products", logger)

    df["_ingested_at"] = datetime.now(timezone.utc).isoformat()

    out_dir  = bronze_dir / "products"
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / "data.parquet"
    df.to_parquet(out_path, index=False)    # ← write succeeds first

    # FR-7.2 — advance watermark ONLY after confirmed write
    new_watermark = str(df["updated_at"].max())
    state.set_watermark(WATERMARK_KEY, new_watermark)
    log_event(logger, "INFO", "products_watermark_advanced",
              new_watermark=new_watermark)

    return out_path

FR-7.1 — StateManager: watermark persistence

src/utils/state.py
class StateManager:
    def __init__(self, state_dir: Path):
        self._dir = state_dir
        self._dir.mkdir(parents=True, exist_ok=True)
        self._watermark_file = self._dir / "watermarks.json"
        self._runs_file      = self._dir / "run_history.jsonl"

    def get_watermark(self, source: str) -> str | None:
        if not self._watermark_file.exists():
            return None
        data = json.loads(self._watermark_file.read_text())
        return data.get(source)

    def set_watermark(self, source: str, value: str) -> None:
        data: dict = {}
        if self._watermark_file.exists():
            data = json.loads(self._watermark_file.read_text())
        data[source] = value
        self._watermark_file.write_text(json.dumps(data, indent=2))

    def record_run(self, metadata: dict) -> None:
        with self._runs_file.open("a") as f:
            f.write(json.dumps(metadata) + "\n")

FR-2.1 — Pydantic schema contracts: OrderRow, CustomerRow, ProductRow

All three schemas live in src/utils/schemas.py. They enforce typed fields, business rules (quantity > 0, valid status values, email format), and are used in the Silver layer to validate every Bronze row.

src/utils/schemas.py — OrderRow
class OrderRow(BaseModel):
    model_config = ConfigDict(coerce_numbers_to_str=False)

    order_id:   str
    customer_id: str
    product_id:  str
    order_date:  date
    quantity:    int
    unit_price:  float
    status:      str

    @field_validator("quantity")
    @classmethod
    def qty_positive(cls, v: int) -> int:
        if v <= 0:
            raise ValueError(f"quantity must be > 0, got {v}")
        return v

    @field_validator("unit_price")
    @classmethod
    def price_positive(cls, v: float) -> float:
        if v < 0:
            raise ValueError(f"unit_price must be >= 0, got {v}")
        return v

    @field_validator("status")
    @classmethod
    def valid_status(cls, v: str) -> str:
        allowed = {"pending", "shipped", "delivered", "cancelled", "returned"}
        if v.lower() not in allowed:
            raise ValueError(f"status '{v}' not in {allowed}")
        return v.lower()

Supporting Infrastructure (both days)

Orchestrator — src/pipeline.py run_one_date()

Wires all stages together. Each stage is wrapped so a failure is captured as FAIL status without polluting subsequent stages. Appends run metadata to state/run_history.jsonl (FR-5.3).

src/pipeline.py — stage runner + run metadata
def run_one_date(date_str: str, config: Config) -> dict:
    logger   = get_logger("novacart", config.logs)
    state    = StateManager(config.state)
    stages:  list[dict] = []

    def stage(name: str, fn):
        t0 = datetime.utcnow()
        try:
            fn()
            stages.append({"stage": name, "status": "OK", ...})
        except Exception as exc:
            stages.append({"stage": name, "status": "FAIL", "error": str(exc), ...})
            raise

    status, error_msg = "SUCCESS", None
    try:
        stage("ingest_orders",    lambda: ingest_orders(...))
        stage("ingest_customers", lambda: ingest_customers(...))
        stage("ingest_products",  lambda: ingest_products(...))
        stage("silver_orders",    lambda: build_silver_orders(...))
        stage("silver_customers", lambda: build_silver_customers(...))
        stage("silver_products",  lambda: build_silver_products(...))
        stage("dim_product",      lambda: build_dim_product(...))
        stage("dim_customer",     lambda: build_dim_customer(...))
        stage("fact_orders",      lambda: build_fact_orders(...))
    except Exception as exc:
        status    = "FAIL"
        error_msg = str(exc)

    metadata = {"date": date_str, "status": status, "error": error_msg, ...}
    state.record_run(metadata)   # FR-5.3
    return metadata

Notes & Observations

ItemObservation
FR-1.5 All-string Bronze The Lab Guide requires Bronze to store everything as str dtype (no type inference). The current ingestion uses pd.read_csv() without dtype=str, so pandas infers numeric types. This is a known Day 3+ concern — the Silver layer is where casting is formally applied. Day 1 & Day 2 tests do not test dtype; all deliverables still pass.
FR-1.6 Partition naming Orders Bronze uses date=YYYY-MM-DD (hive-style). Customers and Products Bronze do not partition by date (no date in their source rhythm). This is consistent with the architecture — customers.json and products.db are not date-scoped files.
FR-1.4 _row_hash The Lab Guide mentions a _row_hash metadata column (FR-1.4). The current implementation adds _source_file and _ingested_at but not a hash. This is not exercised by any Day 1 or Day 2 test — flagged for Day 3+ review.
Day 3+ Scope boundary Silver (src/transform/silver.py), Gold (src/transform/gold.py), and the full 14-scenario test suite are all present in the repo but are formally in scope for Days 3–5. They are not evaluated here.
